@startuml
autonumber
skinparam maxMessageSize 150
skinparam roundcorner 10

box "Pilot Node (F35 Simulator)" #LightBlue
participant "Render Loop\n(main thread 60Hz)" as Main3D
participant "FlightControlComputer\n(g_fcc singoletto)" as FCC
participant "DDS Publisher Thread\n(~20Hz)" as DDSPub
end box

box "DDS Network" #MediumPurple
participant "FastDDS Middleware\n(DDS Domain 0)" as FastDDS
end box

box "Monitor Node (Control Tower)" #LightGreen
participant "DDS Callback Thread\n(FastDDS Asynchronous)" as DDSListener
participant "Monitor Loop\n(main thread)" as MonitorUI
end box

== Inizializzazione ==
Main3D -> FCC: set_initial_state(initial_state)\n[landing_mode=true, system_active=false]
Main3D -> DDSPub: std::thread(dds_publish_thread, writer)
activate DDSPub
DDSPub -> FastDDS: Registra DataWriter QoS (50ms Deadline)

== Flusso di Esecuzione (a regime) ==

par Render Loop (60Hz)
    loop Ogni Frame
        Main3D -> Main3D: Calcola dt (es. 1/60s)
        Main3D -> FCC: get_state()
        activate FCC
        FCC --> Main3D: FlightState (copia pre-step)
        deactivate FCC
        Main3D -> Main3D: HandleInput(Aereo, pilot_input) \n[Raccoglie tasti e aggiorna audio]
        
        Main3D -> FCC: step(pilot_input, dt)
        activate FCC
        FCC -> FCC: law.compute(pilot_input, state, dt) \n[Risolve le Control Law per Normal / Alternate / Direct]
        FCC -> FCC: Integra fisica aerodinamica e propulsiva
        FCC -> FCC: Determina eventuali messaggi di allarme (EICAS)
        FCC --> Main3D: 
        deactivate FCC

        Main3D -> FCC: get_state()
        activate FCC
        FCC --> Main3D: FlightState (stato finale)
        deactivate FCC
        
        Main3D -> Main3D: Draw(Aereo) \n[Renderizza F35 e HUD]
    end
    
else Publisher Sincrono (Monitoraggio FCC)
    loop ~20Hz Target Rate
        DDSPub -> FCC: get_state()
        activate FCC
        FCC --> DDSPub: FlightState (copia)
        deactivate FCC
        DDSPub -> DDSPub: Traduce in "SystemStats"
        DDSPub -> FastDDS: writer->write(&stats)
        DDSPub -> DDSPub: sleep_for(50ms)
    end
    
else Subscriber Asincrono e UI Torre di controllo
    loop Push Data Asincrono
        FastDDS -> DDSListener: on_data_available()
        activate DDSListener
        DDSListener -> DDSListener: Calcola Jitter, Cycle Time, Throughput
        note right: Mutex lock (aereo_mutex) su "shared_aereo"
        DDSListener -> DDSListener: Aggiorna shared_aereo (PlaneData) 
        DDSListener -> DDSListener: Print allarmi formattati a terminale cout
        deactivate DDSListener
    end
    loop Render Loop
        MonitorUI -> MonitorUI: Mutex lock & catch \nlocal_aereo = shared_aereo
        MonitorUI -> MonitorUI: display.Draw(local_aereo) \n[Aggiorna Radar, Horizon, Tapes]
    end
end

@enduml
